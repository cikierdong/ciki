# app.py
# -*- coding: utf-8 -*-
import io, os, re, json, math, hashlib
import numpy as np
import pandas as pd
import streamlit as st

# Optional libs
try:
    import hdbscan  # 密度聚类（更适合不均匀短问法）
    HAS_HDBSCAN = True
except Exception:
    HAS_HDBSCAN = False

from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.cluster import KMeans
from sklearn.metrics.pairwise import cosine_similarity

# 可选句向量：更强同义聚合（首次会下载权重）
USE_ST = False
try:
    from sentence_transformers import SentenceTransformer
    ST_MODEL_NAME = "BAAI/bge-m3"  # 多功能/多语/多粒度
except Exception:
    ST_MODEL_NAME = None

st.set_page_config(page_title="负向问题集生成器", layout="wide")

st.title("📚 负向问题集生成器（上传标注CSV → 标准问/相似问/高频/KB骨架）")
st.caption("基于 TF-IDF/HDBSCAN/K-Means/SimHash，可选句向量（BGE-M3）")

# ========== 0. 上传与列选择 ==========
file = st.file_uploader("上传你的负向问答 CSV（UTF-8/GB 编码均可）", type=["csv"])
if not file:
    st.info("请先上传 CSV。建议包含“用户问题/用户咨询”等列名。")
    st.stop()

# 尝试多编码读取
def read_csv_any(uploaded):
    for enc in ["utf-8-sig","utf-8","gb18030","latin1"]:
        try:
            return pd.read_csv(uploaded, encoding=enc)
        except Exception:
            uploaded.seek(0)
    st.error("无法读取该 CSV，请检查编码/分隔符。")
    st.stop()

df = read_csv_any(file)
df_columns = list(df.columns)

# 识别候选问题列
def guess_q_col(cols):
    pats = ["用户咨询","用户问题","问题","原问题","query","问法","意图问句"]
    for p in pats:
        for c in cols:
            if re.search(p, str(c), flags=re.I):
                return c
    return cols[0]

q_col = st.selectbox("选择【问题列】", options=df_columns, index=max(0, df_columns.index(guess_q_col(df_columns)) if guess_q_col(df_columns) in df_columns else 0))

# ========== 1. 预处理 & 近重复 ==========
def clean_text(x: str) -> str:
    s = str(x)
    s = re.sub(r"\s+", " ", s)
    # 仅移除噪声符号，保留中英文/数字/常见标点
    s = re.sub(r"[^\w\u4e00-\u9fff\.\-_/：:，,。！？!?（）()]+", " ", s)
    return s.strip()

df["_q"] = df[q_col].astype(str).map(clean_text)
df = df[df["_q"].str.len() > 0].copy()

# SimHash 近重复（可选）
use_simhash = st.checkbox("启用近重复合并（SimHash）", value=True,
                          help="合并几乎相同的问法，避免频次被刷高。")
def simhash(s: str, fbits=64):
    # 简单基于字符3-gram的SimHash
    tokens = [s[i:i+3] for i in range(max(1, len(s)-2))]
    v = [0]*fbits
    for tok in tokens:
        h = int(hashlib.md5(tok.encode("utf-8")).hexdigest(), 16)
        for i in range(fbits):
            v[i] += 1 if (h >> i) & 1 else -1
    fp = 0
    for i, val in enumerate(v):
        if val >= 0:
            fp |= (1 << i)
    return fp

def hamming(a, b):
    return bin(a ^ b).count("1")

if use_simhash:
    st.write("→ 正在进行近重复检测…")
    fps = df["_q"].map(lambda s: simhash(s, 64))
    df["__simhash"] = fps
    # 简易桶合并：同哈希直接去重；可扩展阈值（如≤3位差合并）
    df = df.drop_duplicates(subset=["__simhash"]).copy()

st.success(f"清洗后的样本数：{len(df)}")

# ========== 2. 表示学习 ==========
embed_method = st.selectbox("选择表示方法", ["TF-IDF（字符2–4gram）", "SentenceTransformer（BGE-M3，可选）"])
if embed_method.startswith("Sentence") and ST_MODEL_NAME is None:
    st.warning("环境暂不可用 SentenceTransformer，将回落到 TF-IDF。")
    embed_method = "TF-IDF（字符2–4gram）"

if embed_method.startswith("Sentence"):
    USE_ST = True
    @st.cache_resource
    def load_st_model():
        return SentenceTransformer(ST_MODEL_NAME)
    st_model = load_st_model()
    X = st_model.encode(df["_q"].tolist(), normalize_embeddings=True)
else:
    vectorizer = TfidfVectorizer(analyzer="char", ngram_range=(2,4), min_df=1)
    X = vectorizer.fit_transform(df["_q"].tolist())

# ========== 3. 聚类 ==========
clusterer = st.selectbox("聚类算法", ["HDBSCAN（推荐，数据不均匀时）" if HAS_HDBSCAN else "K-Means（备选）", "K-Means（手动K）"])
labels = None

if "HDBSCAN" in clusterer and HAS_HDBSCAN:
    min_cluster_size = st.slider("HDBSCAN: 最小簇大小", 2, 20, 5)
    # HDBSCAN 对密度敏感：稀疏TF-IDF建议metric="cosine"
    cl = hdbscan.HDBSCAN(min_cluster_size=min_cluster_size,
                         metric="cosine" if not USE_ST else "euclidean")
    labels = cl.fit_predict(X if USE_ST else X.toarray())
else:
    default_k = max(2, int(math.sqrt(X.shape[0])))
    K = st.number_input("K-Means: 簇数K", min_value=2, value=default_k, step=1)
    km = KMeans(n_clusters=int(K), n_init=10, random_state=42)
    labels = km.fit_predict(X if USE_ST else X)

df["_cluster"] = labels
valid_mask = df["_cluster"] != -1  # HDBSCAN 的噪声为 -1
noise_cnt = int((df["_cluster"] == -1).sum())
st.write(f"聚类完成：簇数 ≈ {len(set(labels)) - (1 if -1 in labels else 0)}，噪声/离群：{noise_cnt}")

# ========== 4. 标准问（medoid） & 相似问 ==========
def medoid_index(mat, idxs):
    if len(idxs) == 1:
        return idxs[0]
    sub = mat[idxs] if USE_ST else mat[idxs]
    # 计算相似度矩阵
    if USE_ST:
        sim = cosine_similarity(sub)
    else:
        sim = (sub @ sub.T).toarray()
    np.fill_diagonal(sim, 0.0)
    return idxs[int(sim.mean(axis=1).argmax())]

rows = df.reset_index(drop=False).rename(columns={"index":"row_id"})
clusters = {}
for i, lab in enumerate(rows["_cluster"]):
    clusters.setdefault(int(lab), []).append(i)

cluster_rows, mapping_rows, similar_rows = [], [], []

# 向量获取函数
def row_vec(ix):
    return X[ix] if USE_ST else X[ix]

for cid, idxs in clusters.items():
    if cid == -1:  # 噪声跳过，可导出“待人工归并”
        continue
    med = medoid_index(X if USE_ST else X, idxs)
    med_text = rows.iloc[med]["_q"]
    # 簇信息
    cluster_rows.append({
        "cluster_id": int(cid),
        "standard_q_id": int(rows.iloc[med]["row_id"]),
        "standard_q_text": med_text,
        "member_count": len(idxs)
    })
    # 映射与相似问
    center = row_vec(med)
    for idx in idxs:
        vec = row_vec(idx)
        sim = float(cosine_similarity([center],[vec])[0][0]) if USE_ST else float((X[idx] @ center.T).toarray()[0][0])
        mapping_rows.append({
            "row_id": int(rows.iloc[idx]["row_id"]),
            "original_question": rows.iloc[idx]["_q"],
            "cluster_id": int(cid),
            "standard_q_id": int(rows.iloc[med]["row_id"]),
            "similarity": sim
        })
    # 同簇相似问Top
    sims = []
    for idx in idxs:
        if idx == med: continue
        vec = row_vec(idx)
        s = float(cosine_similarity([center],[vec])[0][0]) if USE_ST else float((X[idx] @ center.T).toarray()[0][0])
        sims.append((idx, s))
    sims.sort(key=lambda x: x[1], reverse=True)
    for idx, s in sims[:10]:
        similar_rows.append({
            "standard_q_id": int(rows.iloc[med]["row_id"]),
            "standard_q_text": rows.iloc[med]["_q"],
            "similar_q_id": int(rows.iloc[idx]["row_id"]),
            "similar_q_text": rows.iloc[idx]["_q"],
            "similarity": s
        })

cluster_df = pd.DataFrame(cluster_rows).sort_values("member_count", ascending=False)
mapping_df = pd.DataFrame(mapping_rows)
similar_df = pd.DataFrame(similar_rows)

st.subheader("📊 高频标准问（Top 15）")
st.dataframe(cluster_df.head(15), use_container_width=True)

st.subheader("🔗 标准问 ↔ 相似问（Top 示例）")
st.dataframe(pd.DataFrame(similar_rows).head(20), use_container_width=True)

# ========== 5. KB 稿件骨架（KCS 模板） ==========
kb_cols = ["标准问(标题)","结论/一句话回答","SOP步骤(1->2->3)",
           "前提/限制/适用范围","术语/口径说明(单位/币种/时区)",
           "证据出处(政策/接口字段/版本时间)","相似问(可粘贴或引用)",
           "标签(意图/渠道/人群)"]
kb_df = pd.DataFrame(columns=kb_cols)
kb_df["标准问(标题)"] = cluster_df["standard_q_text"].values
kb_df["相似问(可粘贴或引用)"] = ""

# ========== 6. 导出 ==========
def to_csv_download(df, fname):
    csv = df.to_csv(index=False).encode("utf-8-sig")
    st.download_button("⬇️ 下载 " + fname, data=csv, file_name=fname, mime="text/csv")

st.markdown("---")
col1, col2, col3, col4 = st.columns(4)
with col1: to_csv_download(cluster_df, "clusters.csv")  # 标准问主表
with col2: to_csv_download(similar_df, "similar_questions.csv")  # 标准问-相似问
with col3: to_csv_download(mapping_df, "mapping.csv")  # 原问题映射
with col4: to_csv_download(kb_df, "kb_article_skeleton.csv")  # KB骨架

# ========== 7. 可选：簇主题词（c-TF-IDF） ==========
with st.expander("🧩 生成每簇主题词（可解释，需额外计算）"):
    try:
        from sklearn.feature_extraction.text import TfidfTransformer, CountVectorizer
        # 简化版 c-TF-IDF：将每簇文本拼接为“类文档”，再做 TF-IDF
        docs, labels_order = [], []
        for cid, idxs in clusters.items():
            if cid == -1: continue
            docs.append(" ".join([rows.iloc[i]["_q"] for i in idxs]))
            labels_order.append(cid)
        if len(docs) > 0:
            cv = CountVectorizer(analyzer="char", ngram_range=(2,4), min_df=1)
            counts = cv.fit_transform(docs)
            tfidf = TfidfTransformer().fit_transform(counts)
            vocab = np.array(cv.get_feature_names_out())
            top_terms = []
            for i, cid in enumerate(labels_order):
                row = tfidf[i].toarray().ravel()
                top_idx = row.argsort()[-10:][::-1]
                terms = " ".join(vocab[top_idx])
                top_terms.append({"cluster_id": cid, "top_terms": terms})
            terms_df = pd.DataFrame(top_terms)
            st.dataframe(terms_df, use_container_width=True)
            to_csv_download(terms_df, "cluster_terms.csv")
    except Exception as e:
        st.warning("无法生成主题词：" + str(e))
